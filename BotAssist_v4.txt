--==================== TARGET HELPERS ====================--
local function fallbackPart(model)
	return model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart")
end

local function getPart(model, choice)
	local names = PART_NAMES[choice]
	if names then
		for _, n in ipairs(names) do
			local p = model:FindFirstChild(n)
			if p and p:IsA("BasePart") then
				return p
			end
		end
	end
	return fallbackPart(model)
end

local function closestPart(model, center)
	local best, bestD = nil, math.huge
	for _, p in ipairs(model:GetChildren()) do
		if p:IsA("BasePart") and BODY_SET[p.Name] then
			local sp, on = Camera:WorldToViewportPoint(p.Position)
			if on then
				local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
				if d < bestD then
					best, bestD = p, d
				end
			end
		end
	end
	return best or fallbackPart(model)
end

local function isAlive(model, info)
	local h = info.hum
	return h ~= nil and h.Parent == model and h.Health > 0 and model:IsDescendantOf(workspace)
end

local MODES = {} -- นิยามแต่ละโหมดอยู่ด้านล่าง (MODE DEFINITIONS)

local function isSameTeam(model, info)
	local m = MODES[currentMode]
	if m and m.sameTeam(model, info) then
		return true
	end
	return false
end

local function destroyESP(info)
	if info.hl then
		info.hl:Destroy()
		info.hl = nil
	end
	if info.line then
		info.line:Destroy()
		info.line = nil
	end
	if info.tag then
		info.tag.frame:Destroy()
		info.tag = nil
	end
end

local function unregister(model)
	local info = registry[model]
	if not info then
		return
	end
	registry[model] = nil
	for _, c in ipairs(info.conns) do
		c:Disconnect()
	end
	destroyESP(info)
end

--==================== MODE DEFINITIONS ====================--
-- แต่ละโหมดมี: hook(token) = ต่อ event | scan(token) = สแกนหาเป้า | sameTeam | nameOf | fullName
-- ทุกอย่างที่ทำงานแบบ async รับ token มาเช็ก: ถ้า token ไม่ตรง runToken (= สลับโหมดไปแล้ว) ให้เลิกทันที

-- ---------- BOT: Humanoid ที่ไม่ใช่ผู้เล่น ----------
local function registerHumanoid(hum, token)
	if token ~= runToken or not hum or not hum:IsA("Humanoid") then
		return
	end
	local model = hum.Parent
	if not model or not model:IsA("Model") or registry[model] then
		return
	end
	if hum.Health <= 0 then
		return
	end
	if Players:GetPlayerFromCharacter(model) then
		return -- ผู้เล่น ไม่ใช่บอท
	end
	if BOT_CONTAINER and not model:IsDescendantOf(BOT_CONTAINER) then
		return
	end
	local info = { hum = hum, conns = {} }
	registry[model] = info
	table.insert(
		info.conns,
		model.AncestryChanged:Connect(function()
			if not model:IsDescendantOf(workspace) then
				unregister(model)
			end
		end)
	)
	table.insert(
		info.conns,
		hum.Died:Connect(function()
			unregister(model)
		end)
	)
end

MODES.Bot = {
	label = "BOT",
	singular = "bot",
	plural = "bots",
	sameTeam = function(model)
		-- ทีมของบอท: Attribute "Team" ที่โมเดลบอท (ชื่อทีม หรือ BrickColor)
		local myTeam = LocalPlayer.Team
		if not myTeam then
			return false
		end
		local t = model:GetAttribute("Team")
		if t == nil then
			return false
		end
		return t == myTeam.Name or t == myTeam.TeamColor
	end,
	nameOf = function(model)
		return model.Name
	end,
	fullName = function(model)
		return model:GetFullName()
	end,
	scan = function(token)
		local root = BOT_CONTAINER or workspace
		for _, d in ipairs(root:GetDescendants()) do
			if d:IsA("Humanoid") then
				registerHumanoid(d, token)
			end
		end
	end,
	hook = function(token)
		-- บอทเกิดใหม่ → จับทันที (defer เพื่อให้ Parent พร้อมก่อน)
		table.insert(
			modeConns,
			workspace.DescendantAdded:Connect(function(d)
				if d:IsA("Humanoid") then
					task.defer(registerHumanoid, d, token)
				end
			end)
		)
	end,
}

-- ---------- PLAYER: ผู้เล่นคนอื่น (ข้ามตัวเองเสมอ) ----------
local function registerCharacter(player, char, token)
	if token ~= runToken or player == LocalPlayer or not char or registry[char] then
		return
	end
	local hum = char:FindFirstChildOfClass("Humanoid") or char:WaitForChild("Humanoid", 5)
	-- (WaitForChild รอได้ → เช็กซ้ำหลังรอ: สลับโหมดไปแล้ว / ลงทะเบียนไปแล้ว / ตัวละครหายไปแล้ว)
	if token ~= runToken or not hum or registry[char] or not char:IsDescendantOf(workspace) then
		return
	end
	if hum.Health <= 0 then
		return
	end
	local info = { player = player, hum = hum, conns = {} }
	registry[char] = info
	table.insert(
		info.conns,
		char.AncestryChanged:Connect(function()
			if not char:IsDescendantOf(workspace) then
				unregister(char)
			end
		end)
	)
	table.insert(
		info.conns,
		hum.Died:Connect(function()
			unregister(char)
		end)
	)
end

MODES.Player = {
	label = "PLAYER",
	singular = "player",
	plural = "players",
	sameTeam = function(_, info)
		-- เพื่อนร่วมทีม: ทั้งสองคนต้องมีทีม และเป็นทีมเดียวกัน (แมพไม่มีทีม = ไม่มีใครเป็นเพื่อนร่วมทีม)
		local myTeam = LocalPlayer.Team
		local theirTeam = info.player and info.player.Team
		return myTeam ~= nil and theirTeam ~= nil and myTeam == theirTeam
	end,
	nameOf = function(_, info)
		return info.player.DisplayName
	end,
	fullName = function(_, info)
		return info.player.Name
	end,
	scan = function(token)
		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= LocalPlayer and p.Character and not registry[p.Character] then
				task.spawn(registerCharacter, p, p.Character, token)
			end
		end
	end,
	hook = function(token)
		local function hookPlayer(player)
			if player == LocalPlayer then
				return
			end
			if player.Character then
				task.spawn(registerCharacter, player, player.Character, token)
			end
			table.insert(
				modeConns,
				player.CharacterAdded:Connect(function(char)
					task.spawn(registerCharacter, player, char, token)
				end)
			)
			table.insert(
				modeConns,
				player.CharacterRemoving:Connect(function(char)
					unregister(char)
				end)
			)
		end
		for _, p in ipairs(Players:GetPlayers()) do
			hookPlayer(p)
		end
		table.insert(modeConns, Players.PlayerAdded:Connect(hookPlayer))
		table.insert(
			modeConns,
			Players.PlayerRemoving:Connect(function(p)
				if p.Character then
					unregister(p.Character)
				end
			end)
		)
	end,
}

-- ---------- เริ่ม / หยุดโหมด ----------
-- ลูปสแกนซ้ำทุก 3 วิ (กันบอทที่ Humanoid มาทีหลัง / ตัวละครที่โหลดช้า / ถูกสตรีมกลับเข้ามา)
-- ผูกกับ token: สลับโหมดแล้วลูปของโหมดเก่าจะหยุดเอง
local function startMode()
	local token = runToken
	local m = MODES[currentMode]
	m.hook(token)
	m.scan(token)
	task.spawn(function()
		while token == runToken do
			task.wait(3)
			if token ~= runToken then
				break
			end
			m.scan(token)
		end
	end)
end

-- ลบการทำงานของโหมดปัจจุบันให้เกลี้ยง
local function stopMode()
	runToken = runToken + 1 -- ทำให้ task.spawn / defer / ลูปสแกนของโหมดเก่าเลิกเองทั้งหมด
	for _, c in ipairs(modeConns) do
		c:Disconnect()
	end
	table.clear(modeConns)
	local list = {}
	for model in pairs(registry) do
		table.insert(list, model)
	end
	for _, model in ipairs(list) do
		unregister(model) -- ตัด connection ของเป้า + ลบ Highlight / ป้าย / เส้น
	end
	table.clear(registry)
end

local function hpColor(ratio)
	if ratio < 0.5 then
		return HP_LOW:Lerp(HP_MID, ratio / 0.5)
	end
	return HP_MID:Lerp(HP_FULL, (ratio - 0.5) / 0.5)
end

local function pickFont(name, fallback)
	local ok, f = pcall(function()
		return Enum.Font[name]
	end)
	if ok and f then
		return f, true
	end
	return fallback, false
end

local function new(class, props, parent)
	local o = Instance.new(class)
	for k, v in pairs(props) do
		o[k] = v
	end
	o.Parent = parent

end

end

local function tween(inst, t, props)
	TweenService:Create(inst, TweenInfo.new(t, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props):Play()
end

-- เอฟเฟกต์เมาส์ชี้ (เฉพาะเครื่องที่มีเมาส์ — มือถือไม่ใช้ กันค้างสถานะ hover)
local HAS_MOUSE = UserInputService.MouseEnabled
local function hover(inst, st, onT, offT)
	if not HAS_MOUSE then
		return
	end
	inst.MouseEnter:Connect(function()
		tween(st, 0.1, { Transparency = onT })
	end)
	inst.MouseLeave:Connect(function()
		tween(st, 0.1, { Transparency = offT })
	end)
end

local orderCounter = 0
local function nextOrder()
	orderCounter = orderCounter + 1
	return orderCounter
end

local function isPointer(input)
	return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
end

-- ติดตาม "นิ้วเดียวกัน" เท่านั้น (นิ้วที่กดจอยสติ๊ก/หมุนกล้องจะไม่ไปรบกวน)
local function isSameMove(input, dragInput, dragType)
	if dragType == Enum.UserInputType.Touch then
		return input == dragInput
	end
	return input.UserInputType == Enum.UserInputType.MouseMovement
end
local function isSameEnd(input, dragInput, dragType)
	if dragType == Enum.UserInputType.Touch then
		return input == dragInput
	end
	return input.UserInputType == dragType
end

	local toastToken = 0

	local function toastFade(inst, props)
		local tw = TweenService:Create(inst, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props)
		table.insert(toastTweens, tw)
		tw:Play()
	end
	toast = function(msg)
		toastToken = toastToken + 1
		local my = toastToken
		for _, tw in ipairs(toastTweens) do
			tw:Cancel() -- ยกเลิกการค่อย ๆ จางของข้อความเก่า ไม่งั้นจะมาทับข้อความใหม่
		end

		toastLabel.Text = msg
		toastFrame.Visible = true
		toastFrame.BackgroundTransparency = 0.1
		toastStroke.Transparency = 0.1
		toastLabel.TextTransparency = 0
		task.delay(1.3, function()
			if toastToken ~= my then
				return
			end
			toastFade(toastFrame, { BackgroundTransparency = 1 })
			toastFade(toastStroke, { Transparency = 1 })
			toastFade(toastLabel, { TextTransparency = 1 })
			task.delay(0.4, function()
				if toastToken == my then
					toastFrame.Visible = false
				end
			end)
		end)
	end

		Size = UDim2.fromScale(1, 1),

		BackgroundTransparency = 1,
		BorderSizePixel = 0,
	}, track)
	new("UICorner", { CornerRadius = UDim.new(1, 0) }, trackOn)

		AnchorPoint = Vector2.new(0, 0.5),

		BackgroundColor3 = KNOB_OFF,
		BorderSizePixel = 0,
	}, track)
	new("UICorner", { CornerRadius = UDim.new(1, 0) }, knob)
	local function refresh(animate)
		local on = tbl[key]
		local kp = on and UDim2.new(1, -21, 0.5, 0) or UDim2.new(0, 3, 0.5, 0)
		local kc = on and KNOB_ON or KNOB_OFF
		local tt = on and 0 or 1
		if animate then
			tween(knob, 0.14, { Position = kp, BackgroundColor3 = kc })
			tween(trackOn, 0.14, { BackgroundTransparency = tt })
		else
			knob.Position = kp

			trackOn.BackgroundTransparency = tt
		end

	suffix = suffix or ""
	local r = new("Frame", { Size = UDim2.new(1, -6, 0, 60), BackgroundColor3 = ROW, BorderSizePixel = 0, LayoutOrder = nextOrder() }, page)
	local rs = rowSkin(r)

	lbl.Position = UDim2.fromOffset(12, 5)
	lbl.Size = UDim2.new(1, -90, 0, 22)
	local valueLbl = new("TextLabel", {

		Position = UDim2.new(1, -10, 0, 6),
		Size = UDim2.fromOffset(64, 22),

		Text = tostring(tbl[key]) .. suffix,
		TextColor3 = GOLD_HI,
		Font = Enum.Font.GothamBold,
		TextSize = 12,
	}, r)
	corner(valueLbl, 6)
	stroke(valueLbl, GOLD, 1, 0.78)
	-- พื้นที่แตะกว้าง ๆ สำหรับนิ้วมือถือ
	local hit = new("TextButton", {
		Position = UDim2.new(0, 12, 0, 30),
		Size = UDim2.new(1, -24, 0, 26),
		BackgroundTransparency = 1,
		Text = "",
		AutoButtonColor = false,
	}, r)
	local track = new("Frame", {
		AnchorPoint = Vector2.new(0, 0.5),
		Position = UDim2.fromScale(0, 0.5),
		Size = UDim2.new(1, 0, 0, 6),
		BackgroundColor3 = FIELD,
		BorderSizePixel = 0,
	}, hit)
	new("UICorner", { CornerRadius = UDim.new(1, 0) }, track)
	stroke(track, GOLD, 1, 0.85)
	local a0 = math.clamp((tbl[key] - min) / (max - min), 0, 1)
	local fill = new("Frame", { Size = UDim2.fromScale(a0, 1), BackgroundColor3 = WHITE, BorderSizePixel = 0 }, track)
	new("UICorner", { CornerRadius = UDim.new(1, 0) }, fill)
	gradient(fill, GOLD_LO, GOLD_HI, 0)
	local knob = new("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(a0, 0.5),
		Size = UDim2.fromOffset(17, 17),
		BackgroundColor3 = TEXT,
		BorderSizePixel = 0,
	}, track)
	new("UICorner", { CornerRadius = UDim.new(1, 0) }, knob)
	stroke(knob, GOLD, 2, 0)

	local dragInput, dragType = nil, nil
	local function setFromX(x)
		local a = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
		local v = math.floor(min + (max - min) * a + 0.5)
		tbl[key] = v
		a = (v - min) / (max - min)
		fill.Size = UDim2.fromScale(a, 1)
		knob.Position = UDim2.fromScale(a, 0.5)
		valueLbl.Text = tostring(v) .. suffix
	end
	local function finish()
		if dragInput then
			dragInput, dragType = nil, nil
			page.ScrollingEnabled = true
		end
	end
	hit.InputBegan:Connect(function(input)
		if dragInput == nil and isPointer(input) then
			dragInput, dragType = input, input.UserInputType
			page.ScrollingEnabled = false
			setFromX(input.Position.X)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragInput and isSameMove(input, dragInput, dragType) then
			setFromX(input.Position.X)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if dragInput and isSameEnd(input, dragInput, dragType) then
			finish()
		end
	end)
end

--==================== HOTKEYS (PC) ====================--
-- binds[id] = Enum.KeyCode หรือ false (= ยังไม่ผูกปุ่ม)
local binds = {}
for _, def in ipairs(KEY_DEFS) do
	binds[def.id] = def.default or false
end

local KEY_SHORT = {
	LeftShift = "L-Shift",
	RightShift = "R-Shift",
	LeftControl = "L-Ctrl",
	RightControl = "R-Ctrl",
	LeftAlt = "L-Alt",
	RightAlt = "R-Alt",
	Return = "Enter",
	CapsLock = "Caps",
}
local function keyText(k)
	if not k then
		return "None"
	end
	if KEY_SHORT[k.Name] then
		return KEY_SHORT[k.Name]
	end
	if k.Name:sub(1, 6) ~= "Keypad" then
		-- ตัวอักษร/ตัวเลขตามเลย์เอาต์คีย์บอร์ดของผู้เล่น (ถ้าไม่มี ใช้ชื่อปุ่มตรง ๆ)
		local ok, s = pcall(function()
			return UserInputService:GetStringForKeyCode(k)
		end)
		if ok and type(s) == "string" and s ~= "" then
			return string.upper(s)
		end
	end
	return k.Name
end

local keyPills = {} -- [id] = { pill, stroke }
local listening = nil -- { id, token } เมื่อกำลังรอกดปุ่มใหม่
local listenToken = 0
local keyHeld = false -- ปุ่ม "Lock (Hold)" กำลังถูกกดค้าง

local function refreshKeyPills()
	for _, def in ipairs(KEY_DEFS) do
		local p = keyPills[def.id]
		if p then
			if listening and listening.id == def.id then
				p.pill.Text = "PRESS KEY..."
				p.pill.TextColor3 = GOLD_HI
				p.pill.BackgroundColor3 = Color3.fromRGB(58, 45, 22)
				p.stroke.Transparency = 0
			else
				local k = binds[def.id]
				p.pill.Text = keyText(k)
				p.pill.TextColor3 = k and GOLD_HI or DIM
				p.pill.BackgroundColor3 = FIELD
				p.stroke.Transparency = 0.7
			end
		end
	end
end

stopListening = function()
	if listening then
		listening = nil
		refreshKeyPills()
	end
end

local function startListening(id)
	listenToken = listenToken + 1
	local my = listenToken
	listening = { id = id, token = my }
	refreshKeyPills()
	-- ไม่กดอะไรภายใน 10 วิ → เลิกรอเอง
	task.delay(10, function()
		if listening and listening.token == my then
			stopListening()
		end
	end)
end

-- ผูกปุ่มใหม่: ถ้าปุ่มนั้นถูกใช้อยู่แล้ว → สลับกัน (อีกฟังก์ชันจะได้ปุ่มเดิมของฟังก์ชันนี้)
local function setBind(id, kc)
	local old = binds[id]
	if kc then
		for otherId, k in pairs(binds) do
			if otherId ~= id and k == kc then
				binds[otherId] = old or false
			end
		end
	end
	binds[id] = kc or false
	keyHeld = false
	refreshKeyPills()
end

-- เรียกตอนกำลังรอปุ่ม: Esc = ยกเลิก | Backspace/Delete = ล้างปุ่ม | ปุ่มอื่น = ผูกเลย
local function captureKey(input)
	local kc = input.KeyCode
	local id = listening.id
	if kc == Enum.KeyCode.Escape then
		stopListening()
	elseif kc == Enum.KeyCode.Backspace or kc == Enum.KeyCode.Delete then
		setBind(id, nil)
		stopListening()
	elseif kc ~= Enum.KeyCode.Unknown then
		setBind(id, kc)
		stopListening()
	end
end

local function resetBinds()
	for _, def in ipairs(KEY_DEFS) do
		binds[def.id] = def.default or false
	end
	keyHeld = false
	stopListening()
	refreshKeyPills()
	toast("Keys reset")
end

	end
end

--==================== FLOATING BUTTONS ====================--
-- โครงสร้างปุ่มลอย (กันบั๊กเด้งขึ้นเอง):
--   holder  = กรอบโปร่งใส ใช้ "ย้ายตำแหน่ง" อย่างเดียว (ไม่มี UIScale เลย)
--   └ btn   = ตัวปุ่มจริง อยู่กลาง holder + มี UIScale ไว้ย่อ/ขยาย (เอฟเฟกต์กด + ปรับขนาด)
-- ตำแหน่งเก็บเป็น "สัดส่วนหน้าจอ" (0-1) → ไม่เพี้ยนตอนปรับขนาดปุ่ม/หมุนจอ
local floaters = {}
local DRAG_THRESHOLD = 10 -- px ที่นิ้วต้องขยับก่อนนับว่าเป็น "ลาก" (ไม่งั้นนับเป็นแตะ)

local function applyFloater(f)
	local screen = gui.AbsoluteSize
	if screen.X <= 0 or screen.Y <= 0 then
		return
	end
	local half = f.side / 2
	local cx = math.clamp(f.px * screen.X, half, math.max(half, screen.X - half))
	local cy = math.clamp(f.py * screen.Y, half, math.max(half, screen.Y - half))
	f.px, f.py = cx / screen.X, cy / screen.Y
	f.holder.Position = UDim2.fromScale(f.px, f.py)
end

-- แตะ = onTap | ลาก = ย้ายตำแหน่ง | กดแล้วมีเอฟเฟกต์ยุบ
-- (ติดตามนิ้วเดียวกันตลอด → นิ้วที่กดจอยสติ๊กไม่ทำให้แตะพลาด)
local function makeFloater(holder, btn, base, startPos, onTap)
	local f = {
		holder = holder,
		base = base,
		side = base,
		px = startPos.X,
		py = startPos.Y,
		uiScale = new("UIScale", { Scale = 1 }, btn),
		press = 1,
		target = 1,
	}
	table.insert(floaters, f)
	local dragInput, dragType, moved = nil, nil, false
	local startPointer, grabPointer, grabPos

	local function finish()
		if not dragInput then
			return
		end
		local wasMoved = moved
		dragInput, dragType = nil, nil
		f.target = 1
		if not wasMoved then
			onTap()
		end
	end

	btn.InputBegan:Connect(function(input)
		if dragInput == nil and isPointer(input) then
			dragInput, dragType, moved = input, input.UserInputType, false
			startPointer = Vector2.new(input.Position.X, input.Position.Y)
			f.target = 0.92
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragInput and isSameMove(input, dragInput, dragType) then
			local p = Vector2.new(input.Position.X, input.Position.Y)
			if not moved and (p - startPointer).Magnitude > DRAG_THRESHOLD then
				-- เริ่มลากตรงนี้: จำจุดนิ้ว + ตำแหน่งปุ่ม ณ ตอนนี้ → ปุ่มไม่กระโดดตอนเริ่มลาก
				moved = true
				grabPointer = p
				grabPos = Vector2.new(f.px, f.py)
			end
			if moved then
				local screen = gui.AbsoluteSize
				local d = p - grabPointer
				f.px = grabPos.X + d.X / math.max(screen.X, 1)
				f.py = grabPos.Y + d.Y / math.max(screen.Y, 1)
				applyFloater(f)
			end
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if dragInput and isSameEnd(input, dragInput, dragType) then
			finish()
		end
	end)
	applyFloater(f)
	return f
end

--==================== AIM ====================--
local updateAim
do
	local rayParams = RaycastParams.new()
	rayParams.FilterType = Enum.RaycastFilterType.Exclude
	rayParams.RespectCanCollide = true

	local function isVisible(part, model, camPos)
		local filter = { model }
		if LocalPlayer.Character then
			table.insert(filter, LocalPlayer.Character)
		end
		rayParams.FilterDescendantsInstances = filter
		return workspace:Raycast(camPos, part.Position - camPos, rayParams) == nil
	end

	-- คืนค่า (part, score) ถ้าเป้านี้ล็อกได้
	local function candidate(model, info, camPos, center, needFOV)
		if not isAlive(model, info) then
			return nil
		end
		if S.TeamCheck and isSameTeam(model, info) then
			return nil
		end
		local part
		if S.AimPart == "Closest" then
			part = closestPart(model, center)
		else
			part = getPart(model, S.AimPart)
		end
		if not part then
			return nil
		end
		local dist = (part.Position - camPos).Magnitude
		if dist > S.MaxAimDistance then
			return nil
		end
		local sp, on = Camera:WorldToViewportPoint(part.Position)
		if not on then
			return nil
		end
		local d2 = (Vector2.new(sp.X, sp.Y) - center).Magnitude
		if needFOV and d2 > S.FOVRadius then
			return nil
		end
		if S.WallCheck and not isVisible(part, model, camPos) then
			return nil
		end
		return part, (S.Priority == "Distance") and dist or d2
	end

	local function aimAt(pos, dt)
		local cf = Camera.CFrame
		local goal = CFrame.lookAt(cf.Position, pos)
		if S.Smoothness <= 0 then
			Camera.CFrame = goal
		else
			local base = 1 / (S.Smoothness + 1)
			local alpha = 1 - (1 - base) ^ (dt * 60)
			Camera.CFrame = cf:Lerp(goal, alpha)
		end
	end

	updateAim = function(dt)
		fov.Visible = S.ShowFOV and S.AimEnabled
		fov.Size = UDim2.fromOffset(S.FOVRadius * 2, S.FOVRadius * 2)
		lockHolder.Visible = S.FloatingToggle and S.AimEnabled

		lockActive = S.AimEnabled and (lockToggled or keyHeld or (S.HoldRMB and rmbDown))
		if not lockActive then
			currentTarget = nil
			hasTarget = false
			return
		end

		local camPos = Camera.CFrame.Position
		local center = Camera.ViewportSize / 2
		local part

		if S.Sticky and currentTarget and registry[currentTarget] then
			part = candidate(currentTarget, registry[currentTarget], camPos, center, false)
		end
		if not part then
			currentTarget = nil
			local bestScore = math.huge
			for model, info in pairs(registry) do
				local p, score = candidate(model, info, camPos, center, true)
				if p and score < bestScore then
					bestScore = score
					part = p
					currentTarget = model
				end
			end
		end

		hasTarget = part ~= nil
		if part then
			aimAt(part.Position, dt)
		end
	end
end

--==================== STATUS (แท็บ Finder + แท็บ Mode) ====================--
local function updateStatus()
	local m = MODES[currentMode]
	if not m then
		return
	end
	local total, alive, noPart, nearest = 0, 0, 0, math.huge
	local origin = Camera.CFrame.Position
	for model, info in pairs(registry) do
		total = total + 1
		if isAlive(model, info) then
			alive = alive + 1
			local r = fallbackPart(model)
			if r then
				nearest = math.min(nearest, (r.Position - origin).Magnitude)
			else
				noPart = noPart + 1
			end
		end
	end
	statusLabel.Text = string.format(
		"Tracked %s: %d (alive %d)\nNo body part: %d\nNearest: %s\nStreamingEnabled: %s\n(If true, far-away %s may not be loaded on the client)",
		m.plural,
		total,
		alive,
		noPart,
		(nearest < math.huge) and (math.floor(nearest) .. " studs") or "-",
		tostring(workspace.StreamingEnabled),
		m.plural
	)
	modeInfo.Text = string.format("Active mode: %s\nTracked: %d  (alive %d)", m.label, total, alive)
end

setMode = function(name)
	if not MODES[name] or name == currentMode then
		return
	end
	local first = (currentMode == nil)

	if currentMode then
		-- (1) จำค่า Team Check ของโหมดเก่าไว้ (แต่ละโหมดมีค่าของตัวเอง)
		modeTeam[currentMode].Aim = S.TeamCheck
		modeTeam[currentMode].ESP = S.ESP.TeamCheck
		-- (2) ลบการทำงานของโหมดเก่าให้เกลี้ยง: connection / ลูปสแกน / ESP / Highlight / ป้าย / เส้น / เป้าทั้งหมด
		stopMode()
	end

	-- (3) เคลียร์สถานะล็อก → ไม่มีเป้าเก่าค้าง และไม่ล็อกต่อโดยไม่ได้ตั้งใจ
	lockToggled, lockActive, hasTarget = false, false, false
	keyHeld = false
	currentTarget = nil
	stopListening()

	-- (4) เข้าโหมดใหม่
	currentMode = name
	S.TeamCheck = modeTeam[name].Aim
	S.ESP.TeamCheck = modeTeam[name].ESP
	syncToggles() -- สวิตช์ Team Check ในเมนูขยับตามค่าของโหมดใหม่
	startMode()

	applyModeUI(not first)
	updateStatus()
	if not first then
		toast("Mode: " .. MODES[name].label)
	end
end

setMode(START_MODE)

--==================== MAIN LOOP ====================--
pcall(function()
	RunService:UnbindFromRenderStep("AssistHub")
end)
local statusTimer = 0
RunService:BindToRenderStep("AssistHub", Enum.RenderPriority.Camera.Value + 1, function(dt)
	updateFloaters(dt)
	if not Camera then
		return
	end
	updateESP()
	updateAim(dt)

	-- สีปุ่ม LOCK + เอฟเฟกต์กะพริบเบา ๆ ตอนเปิดล็อก
	refreshFloat()
	local now = os.clock()
	if lockHolder.Visible then
		if lockActive then
			local pulse = 0.5 + 0.5 * math.sin(now * (hasTarget and 8 or 4))
			floatStroke.Transparency = 0.05 + 0.3 * pulse
			glowStroke.Transparency = 0.55 + 0.35 * pulse
		else
			floatStroke.Transparency = 0.1
			glowStroke.Transparency = 0.9
		end
	end

	pulseListening(now)

	-- ประกายทองวิ่งรอบขอบ (หมุนเฉดสีของเส้นขอบ)
	local spin = (now * 45) % 360
	menuGrad.Rotation = spin
	if panel.Visible then
		cardGrad.Rotation = spin
	end
	if fov.Visible then
		fovGrad.Rotation = spin
	end

	statusTimer = statusTimer + dt
	if statusTimer > 0.5 then
		statusTimer = 0
		updateStatus()
	end
end)

local trackedCount = 0
for _ in pairs(registry) do
	trackedCount = trackedCount + 1
end
print("[AssistHub v4.0 GOLD] loaded — mode:", currentMode, "| tracked:", trackedCount)
